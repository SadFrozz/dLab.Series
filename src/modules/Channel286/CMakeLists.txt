cmake_minimum_required(VERSION 3.22)
project(Channel286 VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(cmake/ModelIdentities.cmake)
configure_file(cmake/ModelIdentities.h.in generated/ModelIdentities.h @ONLY)
option(CHANNEL286_BUILD_PLUGIN "Build JUCE plug-in" ON)
option(CHANNEL286_BUILD_TESTS "Build independent DSP tests" ON)
add_library(channel286_dsp STATIC src/dsp/Engine.cpp)
target_include_directories(channel286_dsp PUBLIC src)
set_target_properties(channel286_dsp PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(channel286_dsp PRIVATE /W4)
else()
  target_compile_options(channel286_dsp PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(CHANNEL286_BUILD_TESTS)
  enable_testing()
  add_executable(dsp_tests tests/DspTests.cpp)
  target_link_libraries(dsp_tests PRIVATE channel286_dsp)
  add_test(NAME dsp_regression COMMAND dsp_tests)
  add_executable(circuit_tests tests/CircuitTests.cpp)
  target_link_libraries(circuit_tests PRIVATE channel286_dsp)
  add_test(NAME circuit_calibration COMMAND circuit_tests)
  add_executable(pilot_tests tests/PilotTests.cpp)
  target_link_libraries(pilot_tests PRIVATE channel286_dsp)
  add_test(NAME pilot_circuit_and_curve COMMAND pilot_tests "${CMAKE_BINARY_DIR}/pilot-curves.csv")
endif()
if(CHANNEL286_BUILD_PLUGIN)
  if(NOT TARGET juce::juce_core)
    set(JUCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../deps/JUCE")
    if(NOT EXISTS "${JUCE_DIR}/CMakeLists.txt")
      message(FATAL_ERROR "Initialize the dLab.Series JUCE submodule first")
    endif()
    add_subdirectory("${JUCE_DIR}" juce)
  endif()
  include(${CMAKE_CURRENT_SOURCE_DIR}/../../../cmake/Branding.cmake)
  set(plugin_formats VST3)
  if(APPLE)
    list(APPEND plugin_formats AU)
  endif()
  juce_add_plugin(Channel286
    COMPANY_NAME "Frozz" BUNDLE_ID "${CHANNEL_286_BUNDLE_ID}"
    PLUGIN_MANUFACTURER_CODE ${CHANNEL_MANUFACTURER_CODE} PLUGIN_CODE ${CHANNEL_286_CODE}
    PRODUCT_NAME "${CHANNEL_286_NAME}" FORMATS ${plugin_formats}
    IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE
    IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
  juce_add_binary_data(Channel286Fonts HEADER_NAME Channel286Fonts.h NAMESPACE Channel286Fonts
    SOURCES ../../../resources/fonts/Channel286/BarlowCondensed-Bold.ttf ../../../resources/fonts/Channel286/BarlowCondensed-ExtraBold.ttf
      ../../../resources/fonts/Channel286/BarlowCondensed-SemiBold.ttf ../../../resources/fonts/Channel286/Barlow-LightItalic.ttf ../../../resources/fonts/Channel286/Barlow-Regular.ttf)
  if(MSVC)
    target_compile_options(Channel286 PRIVATE /utf-8)
  endif()
  target_sources(Channel286 PRIVATE src/PluginProcessor.cpp src/PluginEditor.cpp)
  target_include_directories(Channel286 PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_link_libraries(Channel286 PRIVATE dLabBranding Channel286Fonts channel286_dsp juce::juce_dsp juce::juce_audio_utils
    PUBLIC juce::juce_recommended_config_flags)
  target_compile_definitions(Channel286 PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0
    JUCE_VST3_CAN_REPLACE_VST2=0)
  if(APPLE)
    foreach(format IN ITEMS VST3 AU)
      add_custom_command(TARGET Channel286_${format} POST_BUILD
        COMMAND /usr/bin/codesign --force --deep --sign - "$<TARGET_BUNDLE_DIR:Channel286_${format}>"
        COMMENT "Ad-hoc signing Channel286 ${format}")
    endforeach()
  endif()
  if(CHANNEL286_BUILD_TESTS)
    add_executable(plugin_checks tests/PluginChecks.cpp)
    target_link_libraries(plugin_checks PRIVATE Channel286 channel286_dsp juce::juce_dsp juce::juce_audio_utils)
    add_test(NAME plugin_state_and_render COMMAND plugin_checks "${CMAKE_CURRENT_BINARY_DIR}/gui-preview.png")
    add_executable(factory_identity_checks tests/FactoryIdentityChecks.cpp)
    target_link_libraries(factory_identity_checks PRIVATE juce::juce_core juce_vst3_headers)
    target_compile_definitions(factory_identity_checks PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
    target_include_directories(factory_identity_checks PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
    add_dependencies(factory_identity_checks Channel286_VST3)
    add_test(NAME vst3_factory_identity COMMAND factory_identity_checks "$<TARGET_FILE:Channel286_VST3>")
  endif()
endif()
