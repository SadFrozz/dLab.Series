cmake_minimum_required(VERSION 3.22)
project(dLab.Series VERSION 1.0.0 LANGUAGES C CXX)
if(APPLE)
  enable_language(OBJCXX)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
set(JUCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/deps/JUCE")
if(NOT EXISTS "${JUCE_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "JUCE submodule is missing. Run: git submodule update --init --recursive")
endif()
add_subdirectory("${JUCE_DIR}" "${CMAKE_BINARY_DIR}/juce")
include(cmake/Branding.cmake)
set(DLAB_SUITE_PROJECT ON)
add_subdirectory(src/modules/Channel286)
add_subdirectory(src/modules/Channel376)
include(cmake/Bundle.cmake)

if(WIN32)
 add_library(dlab_installer_ui SHARED resources/windows/installer/InstallerUI.rc)
 set_target_properties(dlab_installer_ui PROPERTIES LINKER_LANGUAGE CXX PREFIX "" OUTPUT_NAME "dLabInstallerUI" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/Installer/$<CONFIG>")
 target_link_options(dlab_installer_ui PRIVATE /NOENTRY)
endif()
